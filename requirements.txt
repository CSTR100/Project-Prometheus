requests==2.34.2

# ── planetary raster I/O ─────────────────────────────────────────────────
# Required to read PDS3 .IMG frames (LRO NAC, HiRISE, CTX) and GeoTIFFs.
# Without it the scripts silently fall back to Pillow, which cannot open
# .IMG at all. The PyPI wheel bundles GDAL 3.10 with the PDS/PDS4/ISIS2/
# ISIS3 drivers, so no separate GDAL or ISIS install is needed.
rasterio==1.4.4

# ── deep learning (optional but recommended) ─────────────────────────────
# Without torch, mk19 degrades to the NumPy scorer and xenarch_pipeline
# ignores use_vae=True. The default PyPI wheel on Linux is the CUDA build
# (~3 GB); for a CPU-only box install it first with:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
torch==2.13.0

# ── optional ─────────────────────────────────────────────────────────────
# PDS3 label parsing. lroc_fetch.py imports it behind a try/except and sets
# HAS_PVL, but does not currently use it. Safe to omit.
pvl==1.3.2
